32// Annotation values by annotation name. A later annotation with the same
33// name replaces the earlier one.
34using Annotations = QMap<QString, QString>;
35
36// One argument of a method or a signal.
37struct Argument {
38 QString type; // D-Bus type signature, like "a{sv}".
39 QString name; // May be empty.
40};
41
42using Arguments = QList<Argument>;
43
44// One method of an interface.
45struct Method {
46 QString name;
47 Arguments input_args;
48 Arguments output_args;
49 Annotations annotations;
50};
51
52// One signal of an interface. Signals only have output arguments.
53struct Signal {
54 QString name;
55 Arguments output_args;
56 Annotations annotations;
57};
58
59// One property of an interface.
60struct Property {
61 enum class Access { kRead, kWrite, kReadWrite };
62
63 QString name;
64 QString type; // D-Bus type signature.
65 Access access = Access::kRead;
66 Annotations annotations;
67};
68
69// One interface. The maps are sorted by member name. Overloaded methods and
70// signals share one name, so they use multi maps.
71struct Interface {
72 QString name;
73 // The interface written back as XML, in the same text form that QtDBus
74 // makes. Generated adaptors store it in their D-Bus Introspection
75 // class info.
76 QString introspection;
77 Annotations annotations;
78 QMultiMap<QString, Method> methods;
79 // Not named "signals" because that word is a Qt keyword macro.
80 QMultiMap<QString, Signal> signal_map;
81 QMap<QString, Property> properties;
82};
83
84// Interfaces by interface name.
85using Interfaces = QMap<QString, QSharedPointer<const Interface>>;
